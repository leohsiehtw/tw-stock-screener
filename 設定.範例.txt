# 台股盤後選股 設定檔（範例）
#
# 怎麼用：
#   1. 一般人不用手動改它：雙擊 setup.bat，回答幾個問題就會自動產生 設定.txt。
#   2. 想手動改的話：把這個檔案複製一份，改名為 設定.txt，再用記事本修改。
#
# 規則：
#   - 一行寫一個設定，格式是「項目=內容」，等號兩邊可以有空白。
#   - 以 # 開頭的行是註解，程式會略過。行尾想加說明，前面要留一個空白再寫 #。
#   - 檔案請存成 UTF-8（記事本另存新檔時，下方「編碼」選 UTF-8）。
#   - 寫錯的話，執行時會告訴你是第幾行、哪裡不對，不會默默用錯的設定。

# 總資金（元）。程式會依這個金額試算每檔候選要買幾股。
# 寫 0 或留空 = 不試算部位。可以寫 1000000，也可以寫 1,000,000。
capital=0

# 每筆交易最多願意虧損的比例，也就是跌到建議停損價時，虧掉總資金的幾 %。
# 可以填 2% 到 5%。例如總資金 1,000,000、風險 2%，單筆最多虧 20,000 元。
risk=2%

# 選股範圍：
#   full      = 掃描全市場（上市＋上櫃＋ETF，約 2300 檔）
#   watchlist = 只掃描下面 watchlist 指定的自選股清單
mode=full

# 自選股清單檔案（mode=watchlist 才會用到）。
# 沒有寫完整路徑的話，是相對於這個設定檔所在的資料夾。
watchlist=自選股.csv

# 結果存放的資料夾。程式會在裡面放 盤後選股_日期.html（網頁）、.csv（Excel 用）等檔案。
# 沒有寫完整路徑的話，是相對於這個設定檔所在的資料夾。
# 也可以寫完整路徑，例如：output_dir=D:\選股結果
output_dir=結果

# ---------- 分享版（選用，預設全部關閉） ----------
# 分享版是另一個檔案「分享版_日期.html」：只列出今天的候選標的與關卡說明，
# 不含總資金、部位試算、自選股清單、各關明細、電腦上的路徑。結果資料夾裡的完整網頁（盤後選股_日期.html）
# 含有你的總資金和自選股，請不要直接分享給別人。
#
# share=1：每次選股另外產生 分享版_日期.html（通過安全檢查才會產生）。只存在你的電腦裡，不會上傳。
share=0

# share_publish=1：選股完成後把分享版公開到 GitHub Pages（網址任何人都打得開）。
#   公開有風險：被別人存檔、截圖、搜尋引擎收錄或轉貼後無法收回；請不要在股票名稱以外放個人資料。
#   需要先安裝 GitHub CLI（gh）並執行 gh auth login 登入。
#   第一次公開一定會先列出會上傳什麼，要你自己輸入 Y 確認；排程執行（run_daily.bat）永遠不會做第一次公開。
#   下架：python screener.py share --config 設定.txt --takedown，然後把 share_publish 改回 0。
share_publish=0

# 請準備一個專用的新儲存庫（裡面只會放 index.html 和 README.md 兩個檔案），帳號要和 gh 登入的帳號一樣。
